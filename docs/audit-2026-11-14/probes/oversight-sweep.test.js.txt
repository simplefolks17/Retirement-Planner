function mount() {
  captured.length = 0;
  let r;
  act(() => { r = create(React.createElement(App)); });
  return { latest: () => captured[captured.length - 1], fire: (fn) => act(() => fn()), unmount: () => act(() => r.unmount()) };

const BUNDLES = ["profile", "spending", "accounts", "spouseAccounts", "ss", "pension", "conversion", "health", "assumptions"];

// Every field object in the setter bundles: { path, field }
function fields(p) {
  const out = [];
  const walk = (obj, path) => {
    for (const [k, v] of Object.entries(obj || {})) {
      if (v && typeof v === "object" && typeof v.set === "function" && "value" in v) out.push({ path: `${path}.${k}`, field: v });
      else if (v && typeof v === "object" && !Array.isArray(v)) walk(v, `${path}.${k}`);
    }
  };
  for (const b of BUNDLES) walk(p[b], b);
  return out;
}
const get = (p, path) => path.split(".").reduce((o, k) => o?.[k], p);

// Top-level numeric props = the "headline" signature.
function sig(p) {
  const s = {};
  for (const [k, v] of Object.entries(p)) if (typeof v === "number" && Number.isFinite(v)) s[k] = v;
  for (const k of ["ssView", "rmdView", "conversionView", "taxView", "planHighlights", "rangeView"]) {
    const o = p[k];
    if (o && typeof o === "object") for (const [kk, vv] of Object.entries(o)) if (typeof vv === "number" && Number.isFinite(vv)) s[`${k}.${kk}`] = vv;
  }
  return s;
}
function diff(a, b, noise) {
  return Object.keys(a).filter(k => !noise.has(k) && b[k] !== undefined && Math.abs(a[k] - b[k]) > 1e-9);
}

function altValue(f, path) {
  const v = f.value;
  if (f.options) {
    const opts = f.options.map(o => (typeof o === "object" ? o.value : o));
    if (path.endsWith("selectedState") || path.endsWith("retirementState")) return "CA";
    return opts.find(o => o !== v);
  }
  if (typeof v === "boolean") return !v;
  const min = Number.isFinite(f.min) ? f.min : 0;
  const max = Number.isFinite(f.max) ? f.max : (v ?? min) * 2 + 100_000;
  const step = f.step || 1;
  if (v === null || v === undefined) return Math.round((min + max) / 2 / step) * step;
  let n = v + 4 * step;
  if (n > max) n = v - 4 * step;
  if (n < min) n = (min + max) / 2;
  return n;
}

const MARRY = (app) => {
  app.fire(() => app.latest().ss.isMarried.set(true));
  app.fire(() => app.latest().profile.filingStatus.set("mfj"));
  app.fire(() => app.latest().ss.spouseCurrentAge.set(32));
  app.fire(() => app.latest().profile.spouseIncome.set(80_000));
};

function sweep(label, setup) {
  log(`\n===== INPUT SWEEP: ${label} =====`);
  const a0 = mount(); setup?.(a0);
  const base = sig(a0.latest());
  const list = fields(a0.latest()).map(x => x.path);
  a0.unmount();
  // Noise: remount and compare (Monte Carlo etc.)
  const a1 = mount(); setup?.(a1);
  const again = sig(a1.latest()); a1.unmount();
  const noise = new Set(Object.keys(base).filter(k => Math.abs(base[k] - (again[k] ?? base[k])) > 1e-9));
  log(`noise keys: ${[...noise].join(",") || "(none)"}; signature keys: ${Object.keys(base).length}`);
  for (const path of list) {
    const app = mount(); setup?.(app);
    const f = get(app.latest(), path);
    const alt = altValue(f, path);
    let err = null;
    try { app.fire(() => get(app.latest(), path).set(alt)); } catch (e) { err = e.message; }
    const after = sig(app.latest());
    const now = get(app.latest(), path)?.value;
    const changed = diff(base, after, noise);
    const head = ["totalAtRet", "yearsSustained", "withdrawalRate", "effectiveExpenses", "rmdTaxBite", "netConversionBenefit"]
      .filter(k => changed.includes(k)).join(",");
    log(`${changed.length === 0 ? "DEAD?" : "     "} ${path}: ${JSON.stringify(f.value)} -> ${JSON.stringify(alt)} (stored ${JSON.stringify(now)})  moved ${changed.length}  [${head}]${err ? " ERR " + err : ""}`);
    app.unmount();
  }
}

function boundsCheck(label, app) {
  const bad = [];
  for (const { path, field } of fields(app.latest())) {
    const v = field.value;
    if (typeof v !== "number") continue;
    if (Number.isFinite(field.min) && v < field.min - 1e-9) bad.push(`${path}=${v} < min ${field.min}`);
    if (Number.isFinite(field.max) && v > field.max + 1e-9) bad.push(`${path}=${v} > max ${field.max}`);
    if (Number.isFinite(field.min) && Number.isFinite(field.max) && field.min > field.max) bad.push(`${path}: min ${field.min} > max ${field.max}`);
  }
  log(`[bounds] ${label}: ${bad.length ? "OUT OF RANGE -> " + bad.join(" | ") : "ok"}`);
}
